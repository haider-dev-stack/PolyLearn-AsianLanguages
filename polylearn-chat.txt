const ALLOWED_ORIGIN = "https://haider-dev-stack.github.io";
const MODEL = "gemini-2.5-flash";

const NAMES: Record<string, string> = {
  japanese: "Japanese",
  mandarin: "Mandarin Chinese",
  cantonese: "Cantonese",
  korean: "Korean",
};
const ROMAN: Record<string, string> = {
  japanese: "romaji",
  mandarin: "pinyin with tone marks",
  cantonese: "Jyutping",
  korean: "revised romanization",
};

const cors = {
  "Access-Control-Allow-Origin": ALLOWED_ORIGIN,
  "Access-Control-Allow-Headers": "content-type, apikey, authorization, x-client-info",
  "Access-Control-Allow-Methods": "POST, OPTIONS",
};
const json = (b: unknown, status = 200) =>
  new Response(JSON.stringify(b), { status, headers: { ...cors, "Content-Type": "application/json" } });

// Simple limit: 12 requests per minute per visitor (resets when the function restarts).
const hits = new Map<string, number[]>();
function limited(ip: string) {
  const now = Date.now();
  const recent = (hits.get(ip) || []).filter((t) => now - t < 60000);
  recent.push(now);
  hits.set(ip, recent);
  return recent.length > 12;
}

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") return new Response("ok", { headers: cors });
  if (req.headers.get("origin") !== ALLOWED_ORIGIN) return json({ error: "forbidden" }, 403);
  if (limited((req.headers.get("x-forwarded-for") || "").split(",")[0])) return json({ error: "rate" }, 429);

  let body: any;
  try { body = await req.json(); } catch { return json({ error: "bad request" }, 400); }

  const lang = String(body.lang);
  if (!NAMES[lang]) return json({ error: "bad request" }, 400);
  const contents = (Array.isArray(body.messages) ? body.messages : []).slice(-10).map((m: any) => ({
    role: m.role === "ai" ? "model" : "user",
    parts: [{ text: String(m.text || "").slice(0, 300) }],
  }));
  if (!contents.length || contents[contents.length - 1].role !== "user") return json({ error: "bad request" }, 400);

  const name = NAMES[lang];
  const system =
    `You are a friendly, patient ${name} conversation partner for a beginner. ` +
    `Reply ONLY in ${name}, in one or two short, simple sentences, and end with a simple question to keep the conversation going. ` +
    `Stay on everyday topics such as greetings, food, travel, hobbies and school. ` +
    `Return JSON with these fields: ` +
    `reply (your ${name} reply), ` +
    `reading (${ROMAN[lang]} of your reply), ` +
    `english (English translation of your reply), ` +
    `correction (if the learner's last message has mistakes, is partly or fully English, or sounds unnatural, explain briefly in simple English and give the corrected ${name} sentence; otherwise an empty string), ` +
    `ok (true only if the learner's last message was entirely in ${name} with no mistakes). ` +
    `The learner's text may come from speech recognition, so English spoken aloud can appear as odd ${name} characters. ` +
    `If it looks like garbled English, say so gently in correction and show what they could say in ${name}. ` +
    `Never ask for personal information such as full names, addresses or phone numbers, and refuse unsafe topics politely.`;

  try {
    const r = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/${MODEL}:generateContent`, {
      method: "POST",
      headers: { "Content-Type": "application/json", "x-goog-api-key": Deno.env.get("GEMINI_API_KEY") || "" },
      body: JSON.stringify({
        systemInstruction: { parts: [{ text: system }] },
        contents,
        generationConfig: {
          maxOutputTokens: 500,
          temperature: 0.7,
          responseMimeType: "application/json",
          thinkingConfig: { thinkingBudget: 0 },
          responseSchema: {
            type: "OBJECT",
            properties: {
              reply: { type: "STRING" },
              reading: { type: "STRING" },
              english: { type: "STRING" },
              correction: { type: "STRING" },
              ok: { type: "BOOLEAN" },
            },
            required: ["reply", "reading", "english", "correction", "ok"],
          },
        },
      }),
    });
    if (r.status === 429) return json({ error: "rate" }, 429);
    if (!r.ok) return json({ error: "upstream" }, 502);
    const data = await r.json();
    const out = JSON.parse(data?.candidates?.[0]?.content?.parts?.[0]?.text || "{}");
    return json({
      reply: String(out.reply || "").slice(0, 400),
      reading: String(out.reading || "").slice(0, 400),
      english: String(out.english || "").slice(0, 400),
      correction: String(out.correction || "").slice(0, 600),
      ok: out.ok === true,
    });
  } catch {
    return json({ error: "server" }, 500);
  }
});
